          color: 0xeeeeee,

        stringGrid.add(
          hString

      racket.add(

          // ======================================================
          // PHONE SWING
          // ======================================================

          if (
            data.type ===
              "swing" &&

            data.player ===
              1
          ) {

            const now =
              Date.now();


            if (
              now -
              lastPhoneSwing <
              300
            ) {

              return;
            }


            lastPhoneSwing =
              now;


            const power =
              Number(
                data.power
              ) || 0;


            console.log(
              "PHONE SWING:",
              power
            );


            // Every phone swing is
            // currently a flat shot.

            handleSwing(
              "FLAT"
            );


            return;
          }


          // ======================================================
          // PHONE MOVEMENT
          // ======================================================

          if (
            data.type ===
              "move" &&

            data.player ===
              1
          ) {

            handlePhoneMove(
              data
            );


            return;
          }

        }

        catch (error) {

          console.error(
            "Invalid phone message:",
            error
          );
        }
      };


    // ============================================================
    // PHONE MOVEMENT
    // ============================================================

    function handlePhoneMove(
      data
    ) {

      if (
        data.type !==
          "move" ||

        data.player !==
          1
      ) {

        return;
      }


      const value =
        Number(
          data.x
        );


      if (
        Number.isFinite(
          value
        )
      ) {

        phoneMove =
          THREE.MathUtils.clamp(
            value,
            -1,
            1
          );
      }
    }

    function updateScoreDisplay() {

      if (
        p1Points >= 3 &&
        p2Points >= 3
      ) {

        if (
          p1Points ===
          p2Points
        ) {

          document.getElementById(
            "p1Score"
          ).textContent =
            "DEUCE";


          document.getElementById(
            "p2Score"
          ).textContent =
            "DEUCE";

        }

        else if (
          p1Points ===
          p2Points + 1
        ) {

          document.getElementById(
            "p1Score"
          ).textContent =
            "ADV";


          document.getElementById(
            "p2Score"
          ).textContent =
            "40";

        }

        else if (
          p2Points ===
          p1Points + 1
        ) {

          document.getElementById(
            "p1Score"
          ).textContent =
            "40";


          document.getElementById(
            "p2Score"
          ).textContent =
            "ADV";
        }

      }

      else {

        document.getElementById(
          "p1Score"
        ).textContent =
          tennisScores[
            Math.min(
              p1Points,
              3
            )
          ];


        document.getElementById(
          "p2Score"
        ).textContent =
          tennisScores[
            Math.min(
              p2Points,
              3
            )
          ];
      }


      document.getElementById(
        "p1Games"
      ).textContent =
        `GAMES: ${p1Games}`;


      document.getElementById(
        "p2Games"
      ).textContent =
        `GAMES: ${p2Games}`;


      let setsText =
        `SETS: P1 [${p1Sets}] - [${p2Sets}] P2`;


      if (
        completedSets.length >
        0
      ) {

        setsText +=
          ` (${completedSets.join(", ")})`;
      }


      document.getElementById(
        "sets-summary"
      ).textContent =
        setsText;
    }


    function checkGameAndSetStatus(
      winner
    ) {

      if (
        winner === 0
      )
        p1Points++;

      else
        p2Points++;


      if (
        (
          p1Points >= 4 &&
          p1Points -
          p2Points >= 2
        ) ||

        (
          p2Points >= 4 &&
          p2Points -
          p1Points >= 2
        )
      ) {

        if (
          p1Points >
          p2Points
        )
          p1Games++;

        else
          p2Games++;


        p1Points =
          0;

        p2Points =
          0;


        setStatus(
          p1Points >
          p2Points
            ? "GAME PLAYER 1!"
            : "GAME PLAYER 2!"
        );


        if (
          (
            p1Games >= 6 &&
            p1Games -
            p2Games >= 2
          ) ||

          (
            p2Games >= 6 &&
            p2Games -
            p1Games >= 2
          )
        ) {

          completedSets.push(
            `${p1Games}-${p2Games}`
          );


          if (
            p1Games >
            p2Games
          )
            p1Sets++;

          else
            p2Sets++;


          p1Games =
            0;

          p2Games =
            0;


          if (
            p1Sets === 3 ||
            p2Sets === 3
          ) {

            state =
              "MATCH_OVER";


            setStatus(
              p1Sets === 3
                ? "🏆 MATCH WON BY PLAYER 1!"
                : "🏆 MATCH WON BY PLAYER 2!"
            );


            updateScoreDisplay();

            return;
          }
        }
      }


      updateScoreDisplay();
    }


    function awardPoint(
      winner
    ) {

      state =
        "POINT_OVER";


      checkGameAndSetStatus(
        winner
      );


      if (
        state ===
        "MATCH_OVER"
      )
        return;


      currentServer =
        currentServer === 0
          ? 1
          : 0;


      setTimeout(
        prepareServe,
        1500
      );
    }


    // ============================================================
    // SERVE
    // ============================================================

    function prepareServe() {

      state =
        "SERVING";


      bounceCount =
        0;


      lastHitBy =
        currentServer;


      ballSpin.topspin =
        0;


      if (
        currentServer === 0
      ) {

        player1.group.position.set(
          2.0,
          0,
          9.8
        );


        ballPos.set(
          2.0,
          1.2,
          8.8
        );


        setStatus(
          "YOUR SERVE (PRESS START/PHONE)"
        );

      }

      else {

        player2.group.position.set(
          -2.0,
          0,
          -9.8
        );


        ballPos.set(
          -2.0,
          1.2,
          -8.8
        );


        setStatus(
          "AI SERVING..."
        );


        setTimeout(
          executeAIServe,
          1000
        );
      }


      ballVel.set(
        0,
        0,
        0
      );


      ball.position.copy(
        ballPos
      );
    }


    function executeServe() {

      if (
        state !==
          "SERVING" ||

        currentServer !==
          0
      )
        return;


      player1.serveTimer =
        0.45;


      state =
        "IN_RALLY";


      ballVel.set(
        -0.8,
        5.2,
        -13.5
      );


      ballSpin.topspin =
        2.0;


      setStatus(
        "RALLY!"
      );
    }


    function executeAIServe() {

      if (
        state !==
          "SERVING" ||

        currentServer !==
          1
      )
        return;


      player2.serveTimer =
        0.45;


      state =
        "IN_RALLY";


      ballVel.set(
        0.8,
        5.0,
        13.5
      );


      ballSpin.topspin =
        2.0;


      setStatus(
        "RALLY!"
      );
    }


    // ============================================================
    // SWING
    // ============================================================

    function handleSwing(
      shotType = "FLAT"
    ) {

      if (
        state ===
          "SERVING" &&

        currentServer ===
          0
      ) {

        executeServe();

        return;
      }


      if (
        state !==
        "IN_RALLY"
      )
        return;


      player1.swingTimer =
        0.35;


      const distanceZ =
        Math.abs(
          ballPos.z -
          player1.group.position.z
        );


      const distanceX =
        Math.abs(
          ballPos.x -
          player1.group.position.x
        );


      if (
        distanceZ < 1.8 &&

        distanceX < 2.0 &&

        ballPos.y < 3.2 &&

        ballVel.z > 0
      ) {

        lastHitBy =
          0;


        bounceCount =
          0;


        const offset =
          (
            ballPos.x -
            player1.group.position.x
          ) *
          0.95;


        if (
          shotType ===
          "TOPSPIN"
        ) {

          ballVel.set(
            offset * 2.2,
            6.8,
            -15.0
          );


          ballSpin.topspin =
            8.5;


          setStatus(
            "TOPSPIN DRIVE!"
          );

        }

        else if (
          shotType ===
          "SLICE"
        ) {

          ballVel.set(
            offset * 1.4,
            3.2,
            -10.5
          );


          ballSpin.topspin =
            -6.0;


          setStatus(
            "DROP SLICE!"
          );

        }

        else {

          ballVel.set(
            offset * 2.8,
            4.6,
            -14.2
          );


          ballSpin.topspin =
            1.0;


          setStatus(
            "FLAT SMASH!"
          );
        }
      }
    }


    // ============================================================
    // PHYSICS
    // ============================================================

    function updatePhysics(
      delta
    ) {

      if (
        state !==
        "IN_RALLY"
      )
        return;


      const gravity =
        -9.81;


      const magnusForceY =
        -ballSpin.topspin *
        0.85;


      ballVel.y +=
        (
          gravity +
          magnusForceY
        ) *
        delta;


      ballPos.addScaledVector(
        ballVel,
        delta
      );


      if (
        ballPos.y <= 0.16
      ) {

        ballPos.y =
          0.16;


        ballVel.y =
          -ballVel.y *
          (
            ballSpin.topspin < 0
              ? 0.65
              : 0.78
          );


        ballVel.x *=
          0.88;


        ballVel.z *=
          0.88 +
          (
            ballSpin.topspin *
            0.01
          );


        ballSpin.topspin *=
          0.3;


        bounceCount++;


        const isOut =
          Math.abs(
            ballPos.x
          ) > 5.2 ||

          Math.abs(
            ballPos.z
          ) > 11.88;


        if (
          isOut
        ) {

          awardPoint(
            lastHitBy === 0
              ? 1
              : 0
          );

          return;
        }


        if (
          bounceCount >= 2
        ) {

          awardPoint(
            lastHitBy
          );

          return;
        }
      }


      if (
        Math.abs(
          ballPos.z
        ) < 0.12 &&

        ballPos.y <
        0.98
      ) {

        ballVel.z =
          -ballVel.z *
          0.25;


        ballVel.y =
          1.2;


        setStatus(
          "NET FAULT!"
        );
      }


      // ==========================================================
      // AI
      // ==========================================================

      if (
        ballVel.z < 0 &&
        state ===
        "IN_RALLY"
      ) {

        const targetX =
          THREE.MathUtils.clamp(
            ballPos.x,
            -4.0,
            4.0
          );


        const aiMoveDiff =
          targetX -
          player2.group.position.x;


        player2.group.position.x +=
          aiMoveDiff *
          delta *
          3.2;


        if (
          Math.abs(
            ballPos.z -
            player2.group.position.z
          ) < 1.6 &&

          ballPos.y < 3.0
        ) {

          player2.swingTimer =
            0.35;


          lastHitBy =
            1;


          bounceCount =
            0;


          const returnX =
            (
              Math.random() -
              0.5
            ) *
            6.5;


          ballVel.set(
            returnX -
            player2.group.position.x *
            0.3,

            4.5 +
            Math.random() *
            1.8,

            13.0 +
            Math.random() *
            2
          );


          ballSpin.topspin =
            2.0;


          setStatus(
            "AI RETURNED!"
          );
        }
      }


      if (
        Math.abs(
          ballPos.z
        ) > 13.5
      ) {

        awardPoint(
          lastHitBy
        );
      }


      ball.position.copy(
        ballPos
      );


      shadowTarget.position.set(
        ballPos.x,
        0.03,
        ballPos.z
      );


      const shadowScale =
        Math.max(
          0.3,
          1.0 -
          ballPos.y *
          0.12
        );


      shadowTarget.scale.set(
        shadowScale,
        shadowScale,
        shadowScale
      );
    }


    // ============================================================
    // CHARACTER ANIMATION
    // ============================================================

    function animateCharacter(
      player,
      isMoving,
      delta
    ) {

      if (
        isMoving
      ) {

        const stride =
          Math.sin(
            animTime * 14
          ) *
          0.65;


        player.leftLegGroup.rotation.x =
          stride;


        player.rightLegGroup.rotation.x =
          -stride;

      }

      else {

        player.leftLegGroup.rotation.x *=
          0.8;


        player.rightLegGroup.rotation.x *=
          0.8;
      }


      player.torso.position.y =
        1.25 +
        Math.sin(
          animTime * 4
        ) *
        0.02;


      if (
        player.swingTimer >
        0
      ) {

        player.swingTimer -=
          delta;


        const progress =
          (
            0.35 -
            player.swingTimer
          ) /
          0.35;


        const swingAngle =
          Math.sin(
            progress *
            Math.PI
          );


        player.armPivot.rotation.y =
          -swingAngle *
          2.2;


        player.armPivot.rotation.x =
          -swingAngle *
          0.8;


        player.torso.rotation.y =
          swingAngle *
          0.4;

      }

      else if (
        player.serveTimer >
        0
      ) {

        player.serveTimer -=
          delta;


        const progress =
          (
            0.45 -
            player.serveTimer
          ) /
          0.45;


        const serveAngle =
          Math.sin(
            progress *
            Math.PI
          );


        player.armPivot.rotation.x =
          -serveAngle *
          2.5;


        player.leftArmPivot.rotation.x =
          serveAngle *
          1.8;

      }

      else {

        player.armPivot.rotation.set(
          0,
          0,
          0
        );


        player.leftArmPivot.rotation.set(
          0,
          0,
          0
        );


        player.torso.rotation.y =
          0;
      }
    }

    // ============================================================
    // MAIN LOOP
    // ============================================================

    let lastTime =
      performance.now();


    function animate(
      time
    ) {

      requestAnimationFrame(
        animate
      );


      const delta =
        Math.min(
          0.035,
          (
            time -
            lastTime
          ) / 1000
        );


      lastTime =
        time;


      animTime +=
        delta;


      // ==========================================================
      // PLAYER 1 MOVEMENT
      // ==========================================================

      const prevX1 =
        player1.group.position.x;


      /*
        Keyboard:

        -1 = LEFT
         0 = CENTER
        +1 = RIGHT


        Phone:

        -1 = LEFT
         0 = CENTER
        +1 = RIGHT
      */

      let combinedMove =
        moveDir +
        phoneMove;


      combinedMove =
        THREE.MathUtils.clamp(
          combinedMove,
          -1,
          1
        );


      player1.group.position.x =
        THREE.MathUtils.clamp(
          player1.group.position.x +
          combinedMove *
          delta *
          5.5,

          -4.5,
          4.5
        );


      const isP1Moving =
        Math.abs(
          player1.group.position.x -
          prevX1
        ) > 0.001;


      // ==========================================================
      // AI MOVEMENT
      // ==========================================================

      const prevX2 =
        player2.group.position.x;


      const isP2Moving =
        Math.abs(
          player2.group.position.x -
          prevX2
        ) > 0.001;


      // ==========================================================
      // ANIMATION
      // ==========================================================

      animateCharacter(
        player1,
        isP1Moving,
        delta
      );


      animateCharacter(
        player2,
        isP2Moving,
        delta
      );


      // ==========================================================
      // PHYSICS
      // ==========================================================

      updatePhysics(
        delta
      );


      renderer.render(
        scene,
        camera
      );
    }
